cmake_minimum_required(VERSION 3.16)
project(RobustComparison LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(ZLIB REQUIRED)
set(WAR_BUILD "${CMAKE_CURRENT_SOURCE_DIR}/../../build/ninja-vcpkg" CACHE PATH "Existing what-a-relief Release build")
add_executable(compare-native native.cpp)
target_include_directories(compare-native PRIVATE ../../src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(compare-native PRIVATE "${WAR_BUILD}/what-a-relief-core.lib" ${OpenCV_LIBS} ZLIB::ZLIB)
